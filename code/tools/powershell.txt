### 擷取當前目錄下所有python檔(包含子資料夾)
# 請在 Invest 專案根目錄執行；輸出到 archive，不放在 code。
$exportDir = Join-Path (Get-Location) "archive/code_exports"
New-Item -ItemType Directory -Force -Path $exportDir | Out-Null
$exportFile = Join-Path $exportDir ("all_python_code_" + (Get-Date -Format "yyyyMMdd_HHmmss") + ".txt")
Get-ChildItem -Path . -Filter *.py -File -Recurse |
Where-Object {
    $_.FullName -notmatch '\\(\.venv|venv|env|__pycache__|site-packages|build|dist)(\\|$)'
} |
Sort-Object FullName |
ForEach-Object {
    "`r`n"
    "=" * 100
    "檔案：$($_.FullName)"
    "=" * 100
    Get-Content -LiteralPath $_.FullName -Raw -Encoding UTF8
} | Set-Content -LiteralPath $exportFile -Encoding UTF8


###### 列出當前目錄及其子目錄下所有的檔案名稱(巢狀)
tree /F /A | Out-File -FilePath "$env:USERPROFILE\Desktop\Invest_structure.txt" -Encoding utf8
